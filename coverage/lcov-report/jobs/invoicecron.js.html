const express = require("express");
const pool = require("../config/db");
const cron = require("node-cron");
 
const router = express.Router();
 
async function generateInvoices() {
    try {
        console.log("Starting invoice generation...");
 
       
        const clients = await pool.query(`
            SELECT * FROM clients WHERE next_billing_date = CURRENT_DATE
        `);
 
        for (const client of clients.rows) {
            const clientId = client.id;
 
           
            const parcels = await pool.query(`
                SELECT * FROM parcels WHERE clientid = $1 AND (deliverystatus ='delivered' OR deliverystatus ='return')  AND client_invoiceid IS NULL
            `, [clientId]);
 
            if (parcels.rows.length === 0) continue; 
 
       
            let totalCodAmount = 0, totalReturnFee = 0, totalDeliveryFee = 0, totalCodCommission = 0, totalDiscount = 0;
            let invoiceDetails = [];
 
            parcels.rows.forEach(parcel => {
                const codAmount = parseFloat(parcel.codamount) || 0;
                const returnFee = parseFloat(parcel.return_fee) || 0;
                const deliveryFee = parseFloat(parcel.deliveryfee)|| 0;
                const codCommission = (codAmount - deliveryFee) * (parseFloat(client.cod_commission_fee)/100 || 0);
                const discount = parseFloat(parcel.discount)|| 0;
                const subPayAmount = codAmount - (returnFee + deliveryFee + codCommission - discount)|| 0;
 
                totalCodAmount += codAmount;
                totalReturnFee += returnFee;
                totalDeliveryFee += deliveryFee;
                totalCodCommission += codCommission;
                totalDiscount += discount;
 
          
                invoiceDetails.push([parcel.id, codAmount, returnFee, deliveryFee, codCommission, discount, subPayAmount]);
            });
 
            const subPayAmount = totalCodAmount - (totalReturnFee + totalDeliveryFee + totalCodCommission - totalDiscount) || 0;
            const payAmount = subPayAmount - parseFloat(client.credit_balance) || 0;
 
            const formattedNextBillingDate = new Date(client.next_billing_date);
            formattedNextBillingDate.setDate(formattedNextBillingDate.getDate() + parseInt(client.billing_cycle)); 
            const nextBillingDateISO = formattedNextBillingDate.toISOString().split('T')[0];
 
  
            const invoice = await pool.query(`
                INSERT INTO invoice_header (client_id, invoice_date, next_billing_date, total_cod_amount, total_return_fee, total_delivery_fee, total_cod_commission, item_total_discount, item_sub_pay_amount, pay_amount, bulk_key)
                VALUES ($1, CURRENT_DATE, $2, $3, $4, $5, $6, $7, $8, $9, gen_random_uuid()) 
                RETURNING id
            `, [clientId, nextBillingDateISO, totalCodAmount, totalReturnFee, totalDeliveryFee, totalCodCommission, totalDiscount, subPayAmount, payAmount]);
 
            const invoiceId = invoice.rows[0].id;
 
           
            for (const detail of invoiceDetails) {
                await pool.query(`
                    INSERT INTO invoice_details (invoice_id, parcel_id, cod_amount, return_fee, delivery_fee, cod_commission, discount, item_sub_pay_amount)
                    VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
                `, [invoiceId, ...detail]);
            }
 
       
            await pool.query(`
                UPDATE parcels SET client_invoiceid = $1 WHERE clientid = $2 AND client_invoiceid IS NULL
            `, [invoiceId, clientId]);
 
           
            await pool.query(`
                UPDATE clients SET next_billing_date = next_billing_date + billing_cycle,last_billing_date=CURRENT_DATE WHERE id = $1
            `, [clientId]);
 
            console.log(` Invoice ${invoiceId} generated for Client ${clientId}`);
        }
    } catch (err) {
        console.error(" Error generating invoices:", err);
    }
}
 
 
cron.schedule("* * * * *", generateInvoices);
